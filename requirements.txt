# Python dependencies for the HGMS hIgG separation model, plots and UI.
#
# NOTE: `fs3` (the FS³ simulation bindings) is NOT on PyPI. 
# It is a custom-built package for this project and must be installed/built separately.
numpy
pandas
matplotlib

# Compiles model callbacks to machine code so the solver can call them without the GIL.
numba

# Interactive plots (Plotly) + static PDF/PNG export (kaleido, needs a Chrome/Chromium).
plotly
kaleido

# Web UI.
streamlit
